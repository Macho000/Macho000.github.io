Goでのクエリパラメータの取得

Go

Go言語でクエリパラメーターを取得する方法

答えを先に書くと


package main

import (
	"fmt"
	"net/http"
)

func main() {
	http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
		id := r.URL.Query().Get("id")
		fmt.Println("id =>", id)
	})
	http.ListenAndServe(":3000", nil)
}
